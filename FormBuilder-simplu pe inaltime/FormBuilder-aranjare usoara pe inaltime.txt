  <script>
    const components = document.querySelectorAll(".component");
    const centerPanel = document.getElementById("center-panel");

    // Allow dragging components from the left panel
    components.forEach((component) => {
      component.addEventListener("dragstart", (e) => {
        e.dataTransfer.setData("type", e.target.getAttribute("data-type"));
      });
    });

    // Allow dropping in the center panel
    centerPanel.addEventListener("dragover", (e) => {
      e.preventDefault();
    });

    // Handle drop to add new components
    centerPanel.addEventListener("drop", (e) => {
      e.preventDefault();
      const type = e.dataTransfer.getData("type");
      const newElement = createFormElement(type);

      if (newElement) {
        centerPanel.appendChild(newElement);
      }
    });

    // Function to create form elements dynamically
// Add resizing functionality to form elements
function createFormElement(type) {
  const wrapper = document.createElement("div");
  wrapper.className = "form-element";
  wrapper.draggable = true;
  wrapper.style.position = "relative"; // Required for the resize grip

  let content = "";

  switch (type) {
    case "text-field":
      content = '<label>Text Field: </label><input type="text" placeholder="Enter text" />';
      break;
    case "textarea":
      content = '<label>Text Area: </label><textarea placeholder="Enter text"></textarea>';
      break;
    case "checkbox":
      content = '<label><input type="checkbox" /> Checkbox</label>';
      break;
    case "button":
      content = '<button>Submit</button>';
      break;
    default:
      return null;
  }

  wrapper.innerHTML = content;

  // Add drag-and-drop functionality
  wrapper.addEventListener("dragstart", () => {
    wrapper.classList.add("dragging");
  });

  wrapper.addEventListener("dragend", () => {
    wrapper.classList.remove("dragging");
  });

  // Add resize grip
  const resizeGrip = document.createElement("div");
    resizeGrip.style.width = "0"; /* No width for the arrow */
    resizeGrip.style.height = "0"; /* No height for the arrow */
    resizeGrip.style.position = "absolute";
    resizeGrip.style.right = "0"; /* Align to the right border */
    resizeGrip.style.top = "50%"; /* Center vertically */
    resizeGrip.style.transform = "translateY(-50%)"; /* Adjust for vertical centering */
    resizeGrip.style.cursor = "ew-resize"; /* Horizontal resize cursor */
    resizeGrip.style.borderTop = "5px solid transparent"; /* Arrow top edge */
    resizeGrip.style.borderBottom = "5px solid transparent"; /* Arrow bottom edge */
    resizeGrip.style.borderRight = "5px solid #aaa"; /* Arrow pointing left */


  wrapper.appendChild(resizeGrip);

  // Resize logic
  resizeGrip.addEventListener("mousedown", (e) => {
  e.preventDefault();

  const startX = e.clientX;
  const startWidth = parseInt(window.getComputedStyle(wrapper).width, 10);
  function handleMouseMove(event) {
    const width = Math.max(startWidth + event.clientX - startX, 50); // Minimum width of 50px
    wrapper.style.width = `${width}px`;
  }

  function handleMouseUp() {
    document.removeEventListener("mousemove", handleMouseMove);
    document.removeEventListener("mouseup", handleMouseUp);
  }

  document.addEventListener("mousemove", handleMouseMove);
  document.addEventListener("mouseup", handleMouseUp);
});

  return wrapper;
}


    // Rearranging functionality
    centerPanel.addEventListener("dragover", (e) => {
        e.preventDefault();

        const draggingElement = document.querySelector(".dragging");

        // Ensure `draggingElement` is a valid Node
        if (!draggingElement) return;

        const afterElement = getDragAfterElement(centerPanel, e.clientY);

        if (afterElement) {
            centerPanel.insertBefore(draggingElement, afterElement);
        } else {
            centerPanel.appendChild(draggingElement);
        }
        });

    // Function to determine the position for inserting dragged element
    function getDragAfterElement(container, y) {
      const draggableElements = [
        ...container.querySelectorAll(".form-element:not(.dragging)")
      ];

      return draggableElements.reduce(
        (closest, child) => {
          const box = child.getBoundingClientRect();
          const offset = y - box.top - box.height / 2;

          if (offset < 0 && offset > closest.offset) {
            return { offset, element: child };
          } else {
            return closest;
          }
        },
        { offset: Number.NEGATIVE_INFINITY }
      ).element;
    }
