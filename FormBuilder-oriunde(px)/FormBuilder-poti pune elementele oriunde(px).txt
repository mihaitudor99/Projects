  <script>
    const components = document.querySelectorAll(".component");
    const centerPanel = document.getElementById("center-panel");

    // Allow dragging components from the left panel
    components.forEach((component) => {
      component.addEventListener("dragstart", (e) => {
        e.dataTransfer.setData("type", e.target.getAttribute("data-type"));
      });
    });

    // Allow dropping in the center panel
    centerPanel.addEventListener("dragover", (e) => {
      e.preventDefault();
    });

    centerPanel.addEventListener("drop", (e) => {
      e.preventDefault();

      const type = e.dataTransfer.getData("type");
      const newElement = createFormElement(type, e.offsetX, e.offsetY);

      if (newElement) {
        centerPanel.appendChild(newElement);
      }
    });

    // Function to create form elements dynamically
    function createFormElement(type, x, y) {
      const wrapper = document.createElement("div");
      wrapper.className = "form-element";
      wrapper.style.left = `${x}px`;
      wrapper.style.top = `${y}px`;

      let content = "";

      switch (type) {
        case "text-field":
          content = '<label>Text Field: </label><input type="text" placeholder="Enter text" />';
          break;
        case "textarea":
          content = '<label>Text Area: </label><textarea placeholder="Enter text"></textarea>';
          break;
        case "checkbox":
          content = '<label><input type="checkbox" /> Checkbox</label>';
          break;
        case "button":
          content = '<button>Submit</button>';
          break;
        default:
          alert("Unknown component type!");
          return null;
      }

      wrapper.innerHTML = content;

      // Add drag functionality for repositioning
      addDragFunctionality(wrapper);

      return wrapper;
    }

    function addDragFunctionality(element) {
      let isDragging = false;
      let offsetX = 0;
      let offsetY = 0;

      element.addEventListener("mousedown", (e) => {
        isDragging = true;
        offsetX = e.clientX - element.getBoundingClientRect().left;
        offsetY = e.clientY - element.getBoundingClientRect().top;
        element.classList.add("dragging");
      });

      document.addEventListener("mousemove", (e) => {
        if (!isDragging) return;

        const centerPanelRect = centerPanel.getBoundingClientRect();

        // Keep the element within bounds
        const x = Math.min(
          Math.max(e.clientX - centerPanelRect.left - offsetX, 0),
          centerPanelRect.width - element.offsetWidth
        );
        const y = Math.min(
          Math.max(e.clientY - centerPanelRect.top - offsetY, 0),
          centerPanelRect.height - element.offsetHeight
        );

        element.style.left = `${x}px`;
        element.style.top = `${y}px`;
      });

      document.addEventListener("mouseup", () => {
        if (isDragging) {
          isDragging = false;
          element.classList.remove("dragging");
        }
      });
    }
  </script>
